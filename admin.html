Te lo explicamos de forma transparente:

1. Modo Local (Activo ahora mismo): Todos los cambios que hagas aquí ya se guardan automáticamente en tu navegador (LocalStorage). Perfecto para probar, previsualizar y editar al instante sin configurar nada.

Esquema SQL para Supabase (1 Clic)

Copia este script y pégalo en el SQL Editor de tu panel de Supabase para crear las tablas automáticamente:

-- 1. Tabla de Creadores y Nuevos Campos (Bio, Categoría, Redes, Métricas, Portadas de Caja)
CREATE TABLE IF NOT EXISTS creators (
    id TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    photo TEXT NOT NULL,
    "pngPhoto" TEXT DEFAULT '',
    "boxSettings" JSONB DEFAULT NULL,
    color TEXT NOT NULL DEFAULT '#E63946',
    "inBoxes" BOOLEAN DEFAULT true,
    category TEXT DEFAULT 'Lifestyle & Trends',
    bio TEXT DEFAULT '',
    instagram TEXT DEFAULT '',
    tiktok TEXT DEFAULT '',
    youtube TEXT DEFAULT '',
    metrics TEXT DEFAULT '',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Si la tabla creators ya existía previamente, añade las nuevas columnas automáticamente:
ALTER TABLE creators ADD COLUMN IF NOT EXISTS "pngPhoto" TEXT DEFAULT '';
ALTER TABLE creators ADD COLUMN IF NOT EXISTS "boxSettings" JSONB DEFAULT NULL;
ALTER TABLE creators ADD COLUMN IF NOT EXISTS category TEXT DEFAULT 'Lifestyle & Trends';
ALTER TABLE creators ADD COLUMN IF NOT EXISTS bio TEXT DEFAULT '';
ALTER TABLE creators ADD COLUMN IF NOT EXISTS instagram TEXT DEFAULT '';
ALTER TABLE creators ADD COLUMN IF NOT EXISTS tiktok TEXT DEFAULT '';
ALTER TABLE creators ADD COLUMN IF NOT EXISTS youtube TEXT DEFAULT '';
ALTER TABLE creators ADD COLUMN IF NOT EXISTS metrics TEXT DEFAULT '';

-- 2. Tabla de Casos de Éxito
CREATE TABLE IF NOT EXISTS success_cases (
    id TEXT PRIMARY KEY,
    brand TEXT NOT NULL,
    title TEXT NOT NULL,
    meta TEXT NOT NULL,
    photo TEXT NOT NULL,
    bowl TEXT NOT NULL DEFAULT 'bol_mint.png?v=sat_v2',
    spoon TEXT DEFAULT 'cuchara_1.png',
    "spoonSide" TEXT DEFAULT 'spoon-left',
    stain TEXT DEFAULT 'stain-top-right',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 3. Habilitar lectura y escritura pública en base de datos
ALTER TABLE creators ENABLE ROW LEVEL SECURITY;
ALTER TABLE success_cases ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Lectura pública de creadores" ON creators;
CREATE POLICY "Lectura pública de creadores" ON creators FOR SELECT USING (true);

DROP POLICY IF EXISTS "Escritura pública de creadores" ON creators;
CREATE POLICY "Escritura pública de creadores" ON creators FOR ALL USING (true);

DROP POLICY IF EXISTS "Lectura pública de casos" ON success_cases;
CREATE POLICY "Lectura pública de casos" ON success_cases FOR SELECT USING (true);

DROP POLICY IF EXISTS "Escritura pública de casos" ON success_cases;
CREATE POLICY "Escritura pública de casos" ON success_cases FOR ALL USING (true);

-- 4. Bucket de Supabase Storage para Subida de Fotos
INSERT INTO storage.buckets (id, name, public) 
VALUES ('shakersss-media', 'shakersss-media', true)
ON CONFLICT (id) DO UPDATE SET public = true;

-- Políticas de Supabase Storage para el bucket 'shakersss-media'
DROP POLICY IF EXISTS "Lectura pública de imágenes shakersss" ON storage.objects;
CREATE POLICY "Lectura pública de imágenes shakersss" ON storage.objects 
FOR SELECT USING (bucket_id = 'shakersss-media');

DROP POLICY IF EXISTS "Subida pública de imágenes shakersss" ON storage.objects;
CREATE POLICY "Subida pública de imágenes shakersss" ON storage.objects 
FOR INSERT WITH CHECK (bucket_id = 'shakersss-media');

DROP POLICY IF EXISTS "Actualización de imágenes shakersss" ON storage.objects;
CREATE POLICY "Actualización de imágenes shakersss" ON storage.objects 
FOR UPDATE USING (bucket_id = 'shakersss-media');
